cmake_minimum_required(VERSION 3.16)
project(RouteX VERSION 2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(routex src/main.cpp)
target_link_libraries(routex PRIVATE Threads::Threads)
target_compile_options(routex PRIVATE -Wall -Wextra -Wpedantic)
add_executable(demo_backend src/demo_backend.cpp)
target_compile_options(demo_backend PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
 add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:routex> $<TARGET_FILE:demo_backend>)
endif()
